var UPCOMING = ['NS', 'TBD'];
var FINISHED = ['FT', 'AET', 'PEN'];
var LIVE = ['1H', 'HT', '2H', 'ET', 'BT', 'P', 'LIVE', 'INT'];

function $(id) { return document.getElementById(id); }
function esc(s) {
  return String(s == null ? '' : s).replace(/[&<>"']/g, function (c) {
    return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c];
  });
}
function safeUrl(u) { return /^https?:\/\//i.test(u || '') ? u : ''; }
function has(v) { return v !== null && v !== undefined && v !== ''; }

function pad(n) { return (n < 10 ? '0' : '') + n; }
function hhmm(d) { return pad(d.getHours()) + ':' + pad(d.getMinutes()); }
function dayKey(d) { return d.getFullYear() + '-' + pad(d.getMonth() + 1) + '-' + pad(d.getDate()); }
function parseDate(s) { if (!s) return null; var d = new Date(s); return isNaN(d) ? null : d; }
function leagueRank(id) { var i = LEAGUE_ORDER.indexOf(id); return i < 0 ? 99 : i; }

function dayLabel(key, sample) {
  var now = new Date(), t = dayKey(now);
  var tm = new Date(now.getFullYear(), now.getMonth(), now.getDate() + 1);
  var yd = new Date(now.getFullYear(), now.getMonth(), now.getDate() - 1);
  if (key === t) return 'Jodi a';
  if (key === dayKey(tm)) return 'Demen';
  if (key === dayKey(yd)) return 'Yè';
  if (!sample) return key;
  var s = new Intl.DateTimeFormat('fr', { weekday: 'long', day: 'numeric', month: 'long' }).format(sample);
  return s.charAt(0).toUpperCase() + s.slice(1);
}

function ago(d) {
  if (!d) return '';
  var m = Math.floor((Date.now() - d.getTime()) / 60000);
  if (m < 1) return 'kounye a';
  if (m < 60) return m + ' min de sa';
  if (m < 1440) return Math.floor(m / 60) + ' è de sa';
  return Math.floor(m / 1440) + ' jou de sa';
}

function translateAdvice(a) {
  return String(a || '')
    .replace('Combo Double chance', 'Konbinezon doub chans')
    .replace('Combo Winner', 'Konbinezon genyen')
    .replace('Double chance', 'Doub chans')
    .replace('Winner', 'Genyen')
    .replace(' or draw', ' oswa nul')
    .replace('draw or ', 'nul oswa ')
    .replace(' or ', ' oswa ')
    .replace(' and ', ' ak ')
    .replace('goals', 'gòl');
}

function oddsText(o) {
  if (!o) return 'Kòt: pa disponib';
  var parts = [];
  if (has(o.home)) parts.push('1: ' + Number(o.home).toFixed(2));
  if (has(o.draw)) parts.push('X: ' + Number(o.draw).toFixed(2));
  if (has(o.away)) parts.push('2: ' + Number(o.away).toFixed(2));
  if (!parts.length) return 'Kòt: pa disponib';
  return 'Kòt' + (o.bookmaker ? ' (' + o.bookmaker + ')' : '') + ': ' + parts.join('  ·  ');
}

function usablePrediction(p) {
  if (!p) return null;
  if (String(p.advice || '').toLowerCase().indexOf('no predictions available') >= 0) return null;
  return p;
}

function predictionText(p) {
  if (!p || (!has(p.home) && !has(p.draw) && !has(p.away))) return 'Pronostik: pa disponib';
  return 'Pronostik: Kay ' + (p.home || '') + '  ·  Nul ' + (p.draw || '') + '  ·  Deyò ' + (p.away || '');
}

function logo(url) {
  return url ? '<img src="' + esc(safeUrl(url)) + '" alt="" loading="lazy" onerror="this.style.visibility=\'hidden\'">'
             : '<img alt="" style="visibility:hidden">';
}

function teamsRow(m, center) {
  return '<div class="row"><div class="n h">' + esc(m.home) + '</div>' + logo(m.home_logo) +
    '<div class="mid">' + esc(center) + '</div>' + logo(m.away_logo) +
    '<div class="n">' + esc(m.away) + '</div></div>';
}

function matchCard(m) {
  var st = m.status || '';
  var upcoming = UPCOMING.indexOf(st) >= 0, finished = FINISHED.indexOf(st) >= 0, live = LIVE.indexOf(st) >= 0;
  var center = 'vs';
  if ((finished || live) && has(m.score_home) && has(m.score_away)) center = m.score_home + ' - ' + m.score_away;
  else if (upcoming) { var k = parseDate(m.kickoff); if (k) center = hhmm(k); }
  var html = '<div class="card">' + teamsRow(m, center);
  var s = '';
  if (live && m._unconfirmed) s = 'Dènye skò konnen · pa konfime';
  else if (live) s = st === 'HT' ? 'Mi-tan' : 'An kou' + (has(m.elapsed) ? ' · ' + m.elapsed + "'" : '');
  else if (finished && m._old) s = 'Fini · dènye skò konnen (pa konfime)';
  else if (finished) s = 'Fini';
  else if (st === 'PST') s = 'Ranvwaye';
  else if (st === 'CANC') s = 'Anile';
  else if (st === 'SUSP') s = 'Sispann';
  if (s) html += '<div class="status' + (live && !m._unconfirmed ? ' live' : '') + '">' + esc(s) + '</div>';
  if (upcoming) {
    var p = usablePrediction(m.prediction);
    html += '<div class="info">' + esc(oddsText(m.odds)) + '</div>';
    html += '<div class="info">' + esc(predictionText(p)) + '</div>';
    if (p && p.advice) html += '<div class="info">Konsey sistèm nan: ' + esc(translateAdvice(p.advice)) + '</div>';
  }
  return html + '</div>';
}

function renderLeagues(all) {
  var box = $('leagues');
  var names = [];
  all.slice().sort(function (a, b) { return leagueRank(a.league_id) - leagueRank(b.league_id); })
    .forEach(function (m) { if (m.league && names.indexOf(m.league) < 0) names.push(m.league); });
  var html = '<button data-l="" class="' + (state.league === '' ? 'on' : '') + '">Tout</button>';
  names.forEach(function (n) {
    html += '<button data-l="' + esc(n) + '" class="' + (state.league === n ? 'on' : '') + '">' + esc(n) + '</button>';
  });
  box.innerHTML = html;
  Array.prototype.forEach.call(box.querySelectorAll('button'), function (b) {
    b.onclick = function () { state.league = b.getAttribute('data-l'); render(); };
  });
}

function renderMatches() {
  var all = (state.data.matches || []).filter(Boolean);
  $('leagues').style.display = 'flex';
  renderLeagues(all);
  var list = all.filter(function (m) { return !state.league || m.league === state.league; });
  var today = dayKey(new Date());
  function key(m) { var d = parseDate(m.kickoff); return d ? dayKey(d) : '0000-00-00'; }
  list.sort(function (a, b) {
    var da = key(a), db = key(b), pa = da < today, pb = db < today;
    if (pa !== pb) return pa ? 1 : -1;
    if (da !== db) return pa ? (da < db ? 1 : -1) : (da < db ? -1 : 1);
    var r = leagueRank(a.league_id) - leagueRank(b.league_id);
    if (r) return r;
    return (parseDate(a.kickoff) || 0) - (parseDate(b.kickoff) || 0);
  });
  if (!list.length) return '<div class="msg">Pa gen match pou kounye a nan lig sa yo.</div>' + disclaimer();
  var html = '', lastDay = '', lastLeague = '';
  var isLiveNow = function (m) { return LIVE.indexOf(m.status) >= 0 && !m._unconfirmed; };
  var liveNow = list.filter(isLiveNow);
  if (liveNow.length) {
    html += '<h2>🔴 An kou kounye a</h2>';
    liveNow.forEach(function (m) { html += '<h3>' + esc(m.league) + '</h3>' + matchCard(m); });
    list = list.filter(function (m) { return !isLiveNow(m); });
  }
  list.forEach(function (m) {
    var dk = key(m);
    if (dk !== lastDay) { html += '<h2>' + esc(dayLabel(dk, parseDate(m.kickoff))) + '</h2>'; lastDay = dk; lastLeague = ''; }
    if (m.league !== lastLeague) { html += '<h3>' + esc(m.league + (m.country ? ' · ' + m.country : '')) + '</h3>'; lastLeague = m.league; }
    html += matchCard(m);
  });
  return html + disclaimer();
}

function liveAgeMin() {
  if (!state.live || !state.live.updated) return 9999;
  var t = new Date(state.live.updated).getTime();
  return isNaN(t) ? 9999 : (Date.now() - t) / 60000;
}

// Put the latest live scores (live.json) on top of data.json
function applyLive() {
  if (!state.data) return;
  var matches = state.data.matches || [];
  var age = liveAgeMin();
  var ended = ['PST', 'CANC', 'ABD', 'AWD', 'WO', 'SUSP'];
  if (state.live && state.live.fixtures) {
    matches.forEach(function (m) {
      var f = state.live.fixtures[String(m.id)];
      if (!f || !f.status) return;
      var isLive = LIVE.indexOf(f.status) >= 0;
      var isFinal = FINISHED.indexOf(f.status) >= 0 || ended.indexOf(f.status) >= 0;
      if (!isLive && !isFinal) return;       // not started: keep data.json
      if (isLive && age > 25) return;        // too old to trust a "live" status
      m.status = f.status;
      m.elapsed = f.elapsed;
      m.score_home = f.score_home;
      m.score_away = f.score_away;
      m._confirmed = true;
    });
  }
  // A "live" status that nothing confirmed is only an old snapshot of data.json
  var now = Date.now();
  var u = parseDate(state.data.updated);
  var dataAge = u ? (now - u.getTime()) / 60000 : 9999;
  matches.forEach(function (m) {
    if (LIVE.indexOf(m.status) < 0 || m._confirmed) return;
    var k = parseDate(m.kickoff);
    if (k && now - k.getTime() > 3.5 * 3600 * 1000) { m.status = 'FT'; m._old = true; }
    else if (dataAge > 25) { m._unconfirmed = true; }
  });
}

function renderSponsor() {
  var s = state.sponsor, el = $('sponsor');
  if (s && s.active && s.text) {
    el.textContent = 'Sponsò · ' + s.text;
    var u = safeUrl(s.link);
    if (u) el.setAttribute('href', u); else el.removeAttribute('href');
    el.style.display = 'block';
    el.onclick = function () { track('sponsor_click', { sponsor: String(s.text).slice(0, 60) }); };
  } else el.style.display = 'none';
}

function render() {
  ['match', 'picks', 'news'].forEach(function (t) { $('t-' + t).className = state.tab === t ? 'on' : ''; });
  renderSponsor();
  var c = $('content');
  if (!state.data) {
    $('leagues').style.display = 'none';
    $('updated').textContent = state.note;
    c.innerHTML = '<div class="msg">Done yo poko disponib. Verifye entènèt la epi peze Rafrechi.</div>';
    return;
  }
  var scrollTop = c.scrollTop;
  var u = parseDate(state.data.updated);
  var when = u ? new Intl.DateTimeFormat('fr', { day: 'numeric', month: 'short', hour: '2-digit', minute: '2-digit' }).format(u) : '?';
  var liveInfo = '';
  if (liveAgeMin() < 25) {
    var lu = parseDate(state.live.updated);
    if (lu) liveInfo = ' · Skò an dirèk: ' + hhmm(lu);
  }
  $('updated').textContent = (state.note ? state.note + ' · ' : '') + 'Mizajou: ' + when + liveInfo;
  c.innerHTML = state.tab === 'picks' ? renderPicks() : state.tab === 'news' ? renderNews() : renderMatches();
  c.scrollTop = scrollTop;
}

function getJson(url) {
  return fetch(url + '?t=' + Date.now(), { cache: 'no-store' }).then(function (r) {
    if (!r.ok) throw new Error(r.status);
    return r.json();
  });
}

function load(silent) {
  if (!silent) $('updated').textContent = 'Ap chaje...';
  lastLoad = Date.now();
  var safe = function (p) { return p.catch(function () { return null; }); };
  Promise.all([safe(getJson('data.json')), safe(getJson('sponsor.json')), safe(getJson('offers.json')),
               safe(getJson(LIVE_URL))]).then(function (r) {
    if (r[0]) { state.data = r[0]; state.note = ''; }
    else state.note = state.data ? 'Pa gen koneksyon, done ki sove yo' : '';
    state.sponsor = r[1];
    state.offers = r[2];
    state.live = r[3];
    applyLive();
    render();
  });
}
setInterval(function () { if (!document.hidden && state.data) load(true); }, 5 * 60 * 1000);
document.addEventListener('visibilitychange', function () {
  if (!document.hidden && state.data && Date.now() - lastLoad > 120000) load(true);
});

function setTab(t) { state.tab = t; render(); $('content').scrollTop = 0; track('tab_view', { tab: t }); }
$('t-match').onclick = function () { setTab('match'); };

$('t-news').onclick = function () { setTab('news'); };
$('t-blog').onclick = function () { track('blog_click'); window.open(SITE_URL, '_blank', 'noopener'); };

$('content').addEventListener('click', function (e) {
  var a = e.target && e.target.closest ? e.target.closest('a') : null;
  if (!a) return;
  if (a.classList.contains('news')) track('news_click', { source: (a.querySelector('.mt') || {}).textContent || '' });
  else if (a.classList.contains('offer')) track('offer_click', { name: a.textContent.slice(0, 40) });
});

// ---------- install as an app ----------
var deferredPrompt = null;
function isStandalone() {
  return window.matchMedia('(display-mode: standalone)').matches || window.navigator.standalone === true;
}
function showInstall(text, withButton) {
  if (isStandalone()) return;
  try { if (localStorage.getItem('installClosed') === '1') return; } catch (e) {}
  $('installText').textContent = text;
  $('installBtn').style.display = withButton ? 'inline-block' : 'none';
  $('install').style.display = 'flex';
}
window.addEventListener('beforeinstallprompt', function (e) {
  e.preventDefault();
  deferredPrompt = e;
  track('install_prompt_shown');
  showInstall('Enstale Gollakou sou telefòn ou, tankou yon aplikasyon', true);
});
window.addEventListener('appinstalled', function () { track('app_install'); $('install').style.display = 'none'; deferredPrompt = null; });
$('installBtn').onclick = function () {
  if (!deferredPrompt) return;
  track('install_click');
  deferredPrompt.prompt();
  deferredPrompt.userChoice.then(function () { deferredPrompt = null; $('install').style.display = 'none'; });
};
$('installClose').onclick = function () {
  $('install').style.display = 'none';
  try { localStorage.setItem('installClosed', '1'); } catch (e) {}
};
if (/iphone|ipad|ipod/i.test(navigator.userAgent) && !isStandalone()) {
  showInstall('Sou iPhone: peze bouton Pataje, answit "Ajoute sou ekran akèy".', false);
}

if ('serviceWorker' in navigator) {
  window.addEventListener('load', function () { navigator.serviceWorker.register('sw.js').catch(function () {}); });
}

if (isStandalone()) track('app_open_installed');
render();
load();
